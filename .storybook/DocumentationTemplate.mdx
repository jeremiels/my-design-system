import {
  Meta,
  Title,
  Subtitle,
  Description,
  Markdown,
  Primary,
  Controls,
  Stories,
  useOf,
} from "@storybook/addon-docs/blocks";
import { guidelinesMarkdown } from "./dsds";

{/* The page of every component (preview.ts, docs.page). */}

export const Guidelines = () => {
  const { preparedMeta } = useOf("meta");
  const markdown = guidelinesMarkdown(preparedMeta.component);
  return markdown ? <Markdown>{markdown}</Markdown> : null;
};
